Enemy turn
create:
// Create event of oBattle
sequenceLength = 4;
enemySequence = [];

randomize();

for (var i = 0; i < sequenceLength; i++) {
    enemySequence[i] = irandom(2); // 0 = yellow, 1 = green, 2 = orange
}


showIndex = 0;
showTimer = 0;
showDelay = 30;
showingSequence = true;
flashTime = 20;
gapTime   = 10;
state = "flash";

currentColour = -1; // -1 means nothing lit

playerTurn = false;
playerIndex = 0;

step:
//the colour sequences
if (showingSequence) 
{
    showTimer++;

    if (state == "flash") 
	{
        currentColour = enemySequence[showIndex]; // pick which to show

        if (showTimer >= flashTime) 
		{
            currentColour = -1; // turn off
            showTimer = 0;
            state = "gap";
        }
    }
    else if (state == "gap") 
	{
        if (showTimer >= gapTime) 
		{
            showTimer = 0;
            showIndex++;

            if (showIndex >= sequenceLength) 
			{
                // finished showing all
                showingSequence = false;
                playerTurn = true;
                playerIndex = 0;
                currentColour = -1; // off
            } 
			else 
			{
                state = "flash";
            }
        }
    }
}

//checking what colours the player pressed
	function CheckPlayerInput(colour) 
	{
	    if (colour == enemySequence[playerIndex]) 
		{
	        playerIndex++;
        
	        if (playerIndex >= sequenceLength) 
			{
	            // success!
	            playerTurn = false;
				showingSequence = false
				global.enemyTurn = false
				
				room_goto(rmBmyso)
	        }
	    } 
		else 
		{
	        // failure
	        playerTurn = false;
			showingSequence = false
			global.enemyTurn = false
			global.playerHit = true
			
			room_goto(rmBmyso)
	    }
	}
